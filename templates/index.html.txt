    <script>
        function logMessage(msg) {
            const box = document.getElementById('log-box');
            box.innerHTML += `<br>[${new Date().toLocaleTimeString()}] ${msg}`;
            box.scrollTop = box.scrollHeight;
        }

        function startBot() {
            const email = document.getElementById('iq-email').value;
            const password = document.getElementById('iq-password').value;

            if(!email || !password) {
                alert("Please enter your email and password!");
                return;
            }

            logMessage("Connecting to IQ Option...");
            fetch('/start', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ email: email, password: password })
            })
            .then(res => res.json())
            .then(data => {
                if(data.status === 'success') {
                    logMessage("Connected successfully!");
                } else {
                    logMessage("Error: " + data.message);
                }
            });
        }

        function stopBot() {
            logMessage("Bot stopped.");
            document.getElementById('bot-status').innerText = "Stopped";
            document.getElementById('bot-status').style.color = "#f85149";
        }

        // Poll stats every 3 seconds
        setInterval(() => {
            fetch('/stats')
            .then(res => res.json())
            .then(data => {
                document.getElementById('bot-status').innerText = data.status;
                document.getElementById('bot-status').style.color = data.status === 'Running' ? '#238636' : '#f85149';
                document.getElementById('bot-balance').innerText = data.balance;
            });
        }, 3000);
    </script>
